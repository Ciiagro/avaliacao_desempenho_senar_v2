{# Mostra os itens contestados na abertura de um recurso, cada um em sua própria linha (nome do item + motivo). Recursos antigos, sem itens individualizados, mostram o texto livre como antes. #} {% macro itens_abertura(recurso, texto) -%} {%- if recurso.itens_contestados -%} {%- for item in recurso.itens_contestados -%} {{ item.fator.nome }} {{ item.motivo }} {%- endfor -%} {%- else -%} "{{ texto }}" {%- endif -%} {%- endmacro %} {# Resposta do gestor: cada item numa linha própria, igual à abertura do recurso. Sem itens reconhecidos, mostra o texto livre como antes. #} {% macro itens_resposta(recurso, texto) -%} {%- set nomes = [] -%} {%- for item in recurso.itens_contestados -%}{%- if nomes.append(item.fator.nome) -%}{%- endif -%}{%- endfor -%} {%- set blocos = texto | separar_resposta_por_item(nomes) -%} {%- if blocos and blocos[0][0] -%} {%- for titulo, corpo in blocos -%} {%- if titulo -%}{{ titulo }}{%- endif -%} {{ corpo }} {%- endfor -%} {%- else -%} "{{ texto }}" {%- endif -%} {%- endmacro %}
